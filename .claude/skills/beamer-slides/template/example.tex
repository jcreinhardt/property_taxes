% 4. One estimating equation, every symbol named, the coefficient of interest coloured.
\begin{frame}{Exposure design: compare counties by their 2016 SALT exposure}
  \begin{equation*}
    y_{ct} = \alpha_c + \delta_t + \sum_{k \neq 2017} \cred{\beta_k}\, \mathds{1}[t = k] \times \cblue{E_c} + \varepsilon_{ct}
  \end{equation*}
  \vspace{0.5em}
  \begin{wideitemize}
    \item \cblue{$E_c$}: share of 2016 returns deducting more than \$10{,}000 in state and local taxes
    \item \cred{$\beta_k$}: effect in year $k$ relative to 2017
    \item Identifying assumption: without the cap, high- and low-exposure counties trend alike
  \end{wideitemize}
\end{frame}
